# Fernwood Goods: sales 2026

60 orders, 10 products, 4 regions. Pick a range of months, a segment, or search; click a
region's bar in *Revenue by region* to filter everything, and again to clear it. Move over
the revenue line to read a month. **Esc** clears the region and the search; **←** and **→**
turn the table's pages.

```bib
// ── filters: everything below reads these ──
let from = 1
let to = 12
let seg = "All"
let q = ""
let region = state("region", "")
let tone = {North: accent, South: warning, East: success, West: "#a855f7"}

box(dir: "row", gap: 16, align: end) {
  box(width: 200) { from = slider("From month", 1, 12, value: 1, step: 1) }
  box(width: 200) { to = slider("To month", 1, 12, value: 12, step: 1) }
  seg = choice("Segment", ["All", "Consumer", "Business", "Education"], style: segmented)
  q = input("Search", placeholder: "product or order, e.g. SO-1077")
}
// From pushes To along, and To pulls From back, as the artifact's two month pickers do
if changed("From month") && from > to { set("To month", from) }
if changed("To month") && to < from { set("From month", to) }
let lo = min([from, to])
let hi = max([from, to])
let needle = q.trim().toLowerCase()

if key("escape").pressed {
  set("region", "")
  set("Search", "")
  set("orders", {...value("orders", {}), page: 1})
}

box(dir: "row", gap: 8, align: center) {
  text("{data.months[lo - 1]} – {data.months[hi - 1]} 2026", weight: bold)
  if seg != "All" { badge(seg, tone: accent) }
  if needle != "" { badge("“{q.trim()}”", tone: accent) }
  if region != "" {
    badge("Region: {region}", tone: tone[region])
    if button("Clear region", style: ghost).clicked { set("region", "") }
  }
}

let money = v => v >= 10000 ? fmt(v, currency: "USD", compact: true) : fmt(v, 0, currency: "USD")
let rev = o => o.price * o.units
// all filters but the months and the region; then the two on top
let base = data.orders.filter(o => (seg == "All" || o.segment == seg) &&
  (needle == "" || o.product.toLowerCase().includes(needle) || o.id.toLowerCase().includes(needle)))
let anyMonth = base.filter(o => region == "" || o.region == region)
let shown = anyMonth.filter(o => o.m >= lo && o.m <= hi)
let perMonth = range(1, 13).map(m => anyMonth.filter(o => o.m == m))
```

```bib
// ── KPI tiles: value, change against the period just before, a sparkline of all 12 months ──
let len = hi - lo + 1
let before = anyMonth.filter(o => o.m >= lo - len && o.m < lo)
let total = l => sum(l, rev)
let avg = l => l.length == 0 ? 0 : total(l) / l.length
let units = l => sum(l, o => o.units)
let change = (now, prev) => prev > 0 && lo > 1 ? round((now - prev) / prev * 100, 1) : null

grid(columns: 4) {
  stat("Revenue", money(total(shown)), delta: change(total(shown), total(before)), spark: perMonth.map(total))
  stat("Orders", fmt(shown.length), delta: change(shown.length, before.length), spark: perMonth.map(l => l.length))
  stat("Avg order", money(avg(shown)), delta: change(avg(shown), avg(before)), spark: perMonth.map(avg))
  stat("Units", fmt(units(shown)), delta: change(units(shown), units(before)), spark: perMonth.map(units))
}
text(lo > 1 ? "Change against the {len == 1 ? "month" : "{len} months"} before {data.months[lo - 1]}." : "No earlier period to compare January with.", size: sm, color: muted)
```

```bib
// ── revenue by month (move over it) and revenue by region (click a bar to filter) ──
let monthRev = perMonth.map(l => sum(l, rev))
let orderCount = perMonth.map(l => l.length)
let byRegion = data.regions.map(r => ({r, v: sum(base.filter(o => o.m >= lo && o.m <= hi && o.region == r), rev)}))
let topR = max([...byRegion.map(x => x.v), 1])
let allR = max([sum(byRegion, x => x.v), 1])

row {
  card("Revenue by month", subtitle: "Move over the line to read a month") {
    plot(x: [1, 12], y: [0, ceil(max([...monthRev, 1]) / 2000) * 2000], height: 230, xLabel: "month (1 = Jan)") {
      series("Revenue", range(1, 13), monthRev, fill: true, width: 2.5)
      rule(x: lo, tone: muted)
      rule(x: hi, tone: muted)
    }
  }
  card("Revenue by region", subtitle: "Click a bar to filter everything, again to clear") {
    for x in byRegion {
      let share = round(x.v / allR * 100)
      let b = box(id: "region-{x.r}", dir: "row", gap: 10, align: center, padding: [4, 6], radius: 6,
          opacity: region == "" || region == x.r ? 1 : 0.35, hover: {background: fade(tone[x.r], 0.08)},
          tip: "{x.r}: {money(x.v)}, {share}% of revenue") {
        box(width: 60) { text(x.r, size: sm, weight: region == x.r ? bold : regular) }
        box(width: fill, height: 22, radius: 6, background: fade(tone[x.r], 0.12)) {
          layer { rect(0, 0, w * x.v / topR, h, fill: tone[x.r], radius: 6) }
        }
        box(width: 110) { text("{money(x.v)} · {share}%", size: sm, face: mono) }
      }
      if b.clicked {
        set("region", region == x.r ? "" : x.r)
        set("orders", {...value("orders", {}), page: 1})
      }
    }
  }
}
```

```bib
let one = shown.length == 1 ? shown[0] : null
row {
  card("Price vs units, one dot per order", subtitle: "Search an order id to label its dot") {
    plot(x: [0, 180], y: [0, 30], height: 250, xLabel: "price ($)", yLabel: "units") {
      for r in data.regions {
        let os = shown.filter(o => o.region == r)
        dots(r, os.map(o => o.price + (o.units % 3 - 1) * 2), os.map(o => min([o.units, 30])), tone: tone[r], r: 5)
      }
      if one != null { point(one.price, min([one.units, 30]), label: "{one.id}: {one.units} × ${one.price} = {money(rev(one))}") }
    }
  }
  card("Top products") {
    let names = unique(shown.map(o => o.product))
    let ranked = names.map(p => ({label: p, value: sum(shown.filter(o => o.product == p), rev)})).sort((a, b) => b.value - a.value).slice(0, 6)
    if ranked.length == 0 { text("No orders match.", color: muted) }
    else { bars(ranked, unit: "$") }
  }
}
```

```bib
// ── the orders: click a head to sort (again to reverse), ten to a page ──
// newest first until a head is clicked: the table has no starting sort, so the rows come sorted
let rows = shown.slice().sort((a, b) => a.date < b.date ? 1 : a.date > b.date ? -1 : 0).map(o => [
  o.id, o.date, o.product,
  {text: o.region, value: o.region, color: tone[o.region]},
  o.segment,
  {text: fmt(o.price, currency: "USD"), value: o.price},
  o.units,
  {text: money(rev(o)), value: rev(o)},
])

let t = null
card("Orders", subtitle: "{shown.length} orders match") {
  if rows.length == 0 { text("No orders match these filters.", color: muted) }
  else {
    t = table(["Order", "Date", "Product", "Region", "Segment", "Price", "Units", "Revenue"], rows,
      id: "orders", sort: true, page: 10, widths: [1, 1.1, 2.2, 0.9, 1, 0.9, 0.7, 1.1],
      align: ["start", "start", "start", "start", "start", "end", "end", "end"])
  }
}

// a filter that changes the list sends the table back to its first page
let st = {sort: t?.sort, desc: t?.desc ?? false}
if changed("Search") || changed("Segment") || changed("From month") || changed("To month") { set("orders", {...st, page: 1}) }
if t != null && key("left").pressed && t.page > 1 { set("orders", {...st, page: t.page - 1}) }
if t != null && key("right").pressed && t.page < t.pages { set("orders", {...st, page: t.page + 1}) }
```
